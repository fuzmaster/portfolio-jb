import { renderMedia, selectComposition } from '@remotion/renderer';
import { segmentSchema } from './schemas/segment';
import { parseTranscript } from './parsers/srt';

export async function buildClips(
  srtPath: string,
  branding: BrandConfig
) {
  const raw       = await parseTranscript(srtPath);
  const segments  = segmentSchema.parse(raw);   // throws on bad data

  for (const clip of segments) {
    const comp = await selectComposition({
      serveUrl: bundleUrl,
      id: 'PodcastClip',
      inputProps: { clip, branding },
    });

    await renderMedia({
      composition: comp,
      outputLocation: `./dist/${clip.id}.mp4`,
      codec: 'h264',
    });
    console.log(`✓ rendered ${clip.id}`);
  }
}

type EditSignal = 'cut' | 'keep' | 'retime';

interface RetentionSignal {
  timestamp:    number;
  dropOffRate:  number;
  editDecision: EditSignal;
}

export const mapRetentionSignals = (
  points: WatchPoint[]
): RetentionSignal[] =>
  points.map(({ timestamp, retentionPct }) => ({
    timestamp,
    dropOffRate: 1 - retentionPct,
    editDecision: retentionPct < 0.4
      ? 'cut'
      : retentionPct < 0.65 ? 'retime'
      : 'keep',
  }));

import { z } from 'zod';

export const ReelSchema = z.object({
  project:  z.string().min(1),
  artist:   z.array(z.string()),
  ctaText:  z.enum(['LISTEN ON', 'STREAM ON', 'OUT NOW ON']),
  reelType: z.enum(['Driving', 'Artwork', 'Anytime']),
  timing:   z.object({
    clipDuration: z.number().min(15).max(60),
    audioStart:  z.string().regex(/^\d{2}:\d{2}:\d{2}$/),
    syncFootage: z.boolean().default(false),
  }),
});

export type ReelPayload = z.infer<typeof ReelSchema>;

// Validate at runtime before any render call
export const validatePayload = (raw: unknown): ReelPayload =>
  ReelSchema.parse(raw);

interface Card {
  quantity: number;
  name:     string;
}

export const normalizeCards = (rawText: string): Card[] => {
  const lines = rawText.split(/\r?\n/);

  return lines.flatMap((line) => {
    const trimmed = line.trim();
    if (!trimmed) return [];

    // Match: "4 Lightning Bolt" or "1 Black Lotus"
    const match = trimmed.match(/^(\d+)\s+(.+)$/);
    if (!match) return [];

    return [{
      quantity: parseInt(match[1], 10),
      name:     match[2].trim(),
    }];
  });
};

export function gameReducer(
  state: GameState,
  action: GameAction
): GameState {
  switch (action.type) {
    case 'PLAYER_HIT': {
      // Guard: no hits allowed outside PLAYER_TURN
      if (state.gamePhase !== 'PLAYER_TURN') return state;

      const newDeck       = [...state.deck];
      const drawnCard     = newDeck.pop();
      const newPlayerHand  = [...state.playerHand, drawnCard];
      const handValue     = calcHandValue(newPlayerHand);

      return {
        ...state,
        deck:       newDeck,
        playerHand: newPlayerHand,
        gamePhase:  handValue > 21 ? 'BUST' : 'PLAYER_TURN',
      };
    }
    default: return state;
  }
}

import { test, expect } from '@playwright/test';

const PLATFORMS: Record<string, [number, number]> = {
  instagram:  [1080, 1920],
  youtube:    [1920, 1080],
  tiktok:     [1080, 1920],
};

for (const [platform, [w, h]] of Object.entries(PLATFORMS)) {
  test(`crop-safe: ${platform} ${w}x${h}`, async ({ page }) => {
    await page.setViewportSize({ width: w, height: h });
    await page.goto(`/preview?platform=${platform}`);

    const safeZone = page.locator('[data-safe-zone]');
    await expect(safeZone).toBeVisible();
    await expect(safeZone).not.toBeClipped();
  });
}
