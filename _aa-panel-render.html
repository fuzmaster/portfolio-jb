// pure break-even math — no backend, state persisted to localStorage
export function calculateBreakEven(input: VendorInput) {
  const profitPerItem = input.price - input.itemCost;       // margin per unit
  const fixedCosts    = input.tableFee + input.travel + input.lodging;

  // guard the divide-by-zero / unsellable case
  if (profitPerItem <= 0)
    return { ok: false, warning: 'Margin is zero or negative' };

  const breakEvenUnits = Math.ceil(fixedCosts / profitPerItem);
  const salesPerDay    = breakEvenUnits / input.days;
  const salesPerHour   = salesPerDay / input.hoursPerDay;

  const unrealistic = salesPerHour > 12;        // pace sanity check
  return { breakEvenUnits, salesPerDay, salesPerHour, unrealistic };
}
